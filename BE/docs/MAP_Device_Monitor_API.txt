INPUT: stationId, deviceId?, deviceType?, isEnable?

  ratedPowerKw                 <- tag RATED_POWER
  status                       <- tag MOTOR_STATUS     (running|stopped|error|...)

  runtime.instantH/totalH      <- RUNTIME_INSTANT / RUNTIME
  electrical.voltageRs|St|Tr
             .currentA
             .powerFactor
             .frequencyHz
             .powerKw
             .energyKwh
